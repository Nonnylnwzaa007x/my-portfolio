Overview

สื่อการเรียนรู้เชิงโต้ตอบ (Interactive Multimedia) ที่จำลองกลไกฟิสิกส์นิวเคลียร์ให้เข้าใจง่ายผ่านการเล่นเกม โปรเจกต์นี้เน้นหนักไปที่การเขียน Core Logic และจำลองระบบฟิสิกส์ของอนุภาค เพื่อให้การทดลองในเกมมีความถูกต้องตามหลักวิทยาศาสตร์มากที่สุด

Key Features

  • Physics Simulation: เขียนระบบฟิสิกส์จำลองปฏิกิริยาลูกโซ่ของยูเรเนียม (U235)
  • AI & Behaviors: พัฒนาระบบ AI พื้นฐานและพฤติกรรมวัตถุ เช่น การติดตามเป้าหมาย (Homing System)
  • Core Loop: สร้าง Core Gameplay Loop สำหรับมินิเกม เพื่อให้ผู้เล่นเข้าใจกลไกนิวเคลียร์ผ่านการเล่นอย่างเป็นขั้นตอน

Challenges & Solutions

ความซับซ้อนของกลไกลูกโซ่ (Chain Reaction): การคำนวณการชนและการกระจายตัวของอนุภาคหลายร้อยตัวพร้อมกันทำให้เกมกระตุก ผมแก้ไขปัญหาด้วยการจำกัดจำนวนอนุภาคสูงสุดที่จะทำการประมวลผลต่อเฟรม (Time-slicing) และใช้คณิตศาสตร์คำนวณทิศทางฟิสิกส์ล่วงหน้า (Pre-computation) ทำให้ระบบมีเสถียรภาพมากขึ้น

ตัวอย่างโค้ด: Homing System

void Update() {
    if (target != null) {
        // คำนวณเวกเตอร์ทิศทางไปยังเป้าหมาย
        Vector3 direction = (target.position - transform.position).normalized;
        
        // คำนวณองศาการหมุนและใช้ Slerp เพื่อให้หมุนอย่างนุ่มนวล
        Quaternion lookRotation = Quaternion.LookRotation(direction);
        transform.rotation = Quaternion.Slerp(transform.rotation, lookRotation, Time.deltaTime * turnSpeed);
        
        // เคลื่อนที่ไปข้างหน้าตามทิศทางที่หันไป
        transform.position += transform.forward * speed * Time.deltaTime;
    }
}

ตัวอย่างโค้ด: Chain Reaction Physics

ในการจำลองปฏิกิริยาลูกโซ่ (Chain Reaction) ผมใช้ OverlapSphere ในการค้นหาอนุภาค U235 รอบๆ จุดที่เกิดการแตกตัว และสร้างแรงผลัก (ExplosionForce) เพื่อจำลองการกระจายตัวแบบฟิสิกส์ 

private void TriggerFission() {
    Collider[] hitColliders = Physics.OverlapSphere(transform.position, reactionRadius);
    foreach (var hitCollider in hitColliders) {
        AtomBehavior atom = hitCollider.GetComponent<AtomBehavior>();
        
        if (atom != null && atom.isU235) {
            Rigidbody rb = hitCollider.GetComponent<Rigidbody>();
            if (rb != null) {
                // ผลักอนุภาคข้างเคียงให้ออกห่าง
                rb.AddExplosionForce(fissionForce, transform.position, reactionRadius);
            }
            // เรียกให้เกิด Fission ต่อเป็นลูกโซ่
            atom.StartReactionDelay(); 
        }
    }
}

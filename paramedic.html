Overview

เกม VR จำลองสถานการณ์ฉุกเฉินทางการแพทย์ โดยให้ผู้เล่นรับบทบาทเป็นหน่วยกู้ชีพ (Paramedic) เข้าไปช่วยเหลือผู้บาดเจ็บ ณ จุดเกิดเหตุ ในโปรเจกต์นี้ผมได้รับหน้าที่พัฒนาระบบ Gameplay หลัก โดยเน้นที่ ระบบจำลองการกดหน้าอกทำ CPR เสมือนจริง และ ระบบบริหารจัดการสลับฉากเคสผู้ป่วย (Scene & Sub-Scene Management) เพื่อให้ขั้นตอนการเล่นในโลก VR มีความต่อเนื่องและลื่นไหล

Key Features

  • CPR Simulation: จำลองการทำ CPR ปั๊มหัวใจ โดยคำนวณตำแหน่งมือและการทิ้งน้ำหนักตัว พร้อมวัดอัตราความเร็วการกด (BPM) แบบ Real-time
  • Scene Management: วางระบบจัดการฉากแบบ Hub และ Sub-Scenes เพื่อควบคุมการสลับฉากเคสผู้ป่วยอย่างเป็นระบบ และป้องกันการทำงานทับซ้อน

Challenges & Solutions

1. การจำลองการกดหน้าอกทำ CPR ในโลก VR

เนื่องจากอุปกรณ์ VR ไม่มีแรงต้านทานทางกายภาพ (Force Feedback) หากใช้การตรวจจับการชนทั่วไป มือจะกดทะลุตัวผู้ป่วย จึงได้ออกแบบระบบ CPRController โดยตรวจจับตำแหน่งมือให้อยู่บนหน้าอก ร่วมกับการตรวจจับระยะการโน้มตัวของผู้เล่น (ระยะกดต่ำลงของ Headset) เนื่องจากเวลาทำ CPR จริงจะต้องทิ้งน้ำหนักตัวลง พร้อมคำนวณอัตราความเร็วการกด (BPM) แบบ Real-time และใส่ระบบ Debounce เพื่อป้องกันการนับรอบซ้ำ 

// Biomechanical CPR Detection & Real-time BPM Calculation
float currentY = _headAnchor.position.y;
float drop = _baselineY - currentY;

if (drop >= _compressionDepth && !_isCompressing) {
    _isCompressing = true;
    float now = Time.time;
    float elapsed = now - _lastCompressionTime;
    if (elapsed < _minInterval) return; // Debounce ป้องกันนับซ้ำ

    _compressionCount++;
    float bpm = 60f / elapsed;
    _lastCompressionTime = now;

    if (bpm < 200f) {
        OnCompressionSuccess?.Invoke(bpm); // อัปเดตตัวเลขขึ้น UI Text
    }

    if (_compressionCount >= _requiredCompressions) {
        _isCompleted = true;
        OnCPRComplete?.Invoke(); // ผ่านเงื่อนไขกู้ชีพสำเร็จ
    }
}
else if (drop < _compressionDepth * 0.3f && _isCompressing) {
    _isCompressing = false;
    _baselineY = currentY; // รีเซ็ตความสูงเตรียมรับการกดรอบถัดไป
}

2. ระบบจัดการและเปลี่ยนผ่านฉาก (Hub & Sub-Scene Management)

การจัดการเกมที่มีหลายสถานการณ์ในโลก VR จำเป็นต้องควบคุมการโหลดและสลับฉากอย่างแม่นยำ จึงได้พัฒนาระบบ MiniGame3Manager เพื่อทำหน้าที่เป็นศูนย์กลาง ควบคุมการสลับระหว่างฉากหลัก (Hub Scene) และฉากเคสย่อย (Sub-Scenes) พร้อมจัดการสถานะ IsCaseActive เพื่อป้องกันไม่ให้ Interaction หรือ Trigger ของฉากอื่นทำงานซ้อนทับกันในขณะที่ผู้เล่นกำลังทำภารกิจ 

// ระบบเปิด Sub-Scene เคสผู้ป่วย และป้องกันการทำงานทับซ้อนใน VR
private void OpenSubScene(GameObject subScene) {
    if (subScene != null) {
        IsCaseActive = true;
        subScene.SetActive(true);
    }
    if (_pointerManager != null) _pointerManager.SetActive(false);
    
    // ซ่อน Target อื่นในฉากหลักเพื่อป้องกัน Trigger ซ้อนทับ
    if (_hubGreen != null) _hubGreen.SetActive(false);
    if (_hubYellow != null) _hubYellow.SetActive(false);
    if (_hubBlack != null) _hubBlack.SetActive(false);
}

private void ReturnToHub(GameObject currentSubScene) {
    IsCaseActive = false;
    CloseSubScene(currentSubScene);
}

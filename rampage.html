Overview

เกม 2D Action Hack'n Slash การเอาชีวิตรอดในโลกในวันสิ้นโลก ใน Project นี้ผมได้รับหน้าที่หลักในการพัฒนาระบบเกมเพลย์ทั้งหมด (Core Gameplay Systems) ด้วย C# และ Monogame Framework เพื่อให้การควบคุมตัวละครและการต่อสู้มีความลื่นไหลและสนุกสนาน

Key Systems Developed

  • Player System: พัฒนาระบบการเคลื่อนไหว (วิ่ง, กระโดด, พุ่งตัว) และจัดการ State/Animation 2D (Idle, Run, Jump, Attack, Die)
  • Combat System: สร้างระบบโจมตีสกิล และระบบตรวจสอบ Hitbox แบบ Rectangle Bounding Box
  • Enemy & Boss System: เขียนระบบพฤติกรรมของศัตรู (ซอมบี้, หมึก) รวมถึงบอส Malraku ที่มีรูปแบบการโจมตีเฉพาะตัว
  • Environment System: สร้างฉากคัตซีน และระบบ Parallax Scrolling เพื่อให้ฉากหลังเคลื่อนที่อย่างมีมิติ
  • Audio System: ผสานเอฟเฟกต์เสียงต่างๆ ทั้งการฟันดาบ, สกิล, เสียงเดิน, และจังหวะหัวใจเต้นเมื่อ HP ต่ำ

Challenges & Solutions

ระบบ Combat และ Hit Detection: ในเกมแนว 2D Hack'n Slash การต่อสู้ต้องมีความแม่นยำและสัมพันธ์กับเฟรมแอนิเมชัน ผมได้เขียนระบบตรวจสอบการชนด้วย Rectangle Bounding Box ในภาษา C# โดยแยก _playerHitbox สำหรับรับดาเมจ และ _attackHitbox สำหรับการสร้างดาเมจ ซึ่งจะคำนวณและสร้าง Hitbox ขึ้นมาเฉพาะเฟรมแอนิเมชันที่ตัวละครกำลังง้างฟันจริงๆ เท่านั้น เพื่อให้ระบบฟิสิกส์การโจมตีมีความสมจริง

ตัวอย่างโค้ด Logic: Hitbox Detection (C# Monogame)

_playerHitbox = new Rectangle((int)_position.X + 60, (int)_position.Y + 50, 110, 150);

if (_isAttacking) {
    if (!_isFlip)
        _attackHitbox = new Rectangle((int)_position.X + 150, (int)_position.Y + 50, 120, 120);
    else
        _attackHitbox = new Rectangle((int)_position.X - 50, (int)_position.Y + 50, 130, 120);
} else {
    _attackHitbox = new Rectangle(0, 0, 0, 0);
}

if (_attackHitbox.Intersects(enemy._hitbox)) {
    enemy.TakeDamage(_damage);
    _hitSoundInstance.Play();
}

ตัวอย่างโค้ด Logic: Movement & State System

การควบคุมตัวละคร 2D Action สิ่งสำคัญคือความลื่นไหลของระบบฟิสิกส์ (Gravity & Velocity) ควบคู่ไปกับการเปลี่ยน State ตัวละคร โค้ดส่วนนี้เป็นการรับ Input ควบคุมการเดินซ้าย-ขวา การกระโดด และจัดการตำแหน่งบนแกน Y เมื่อตัวละครตกถึงพื้น 

KeyboardState state = Keyboard.GetState();
_isRunning = false;

if (state.IsKeyDown(Keys.D)) {
    _position.X += 400f * (float)gameTime.ElapsedGameTime.TotalSeconds;
    _isRunning = true;
}
if (state.IsKeyDown(Keys.A)) {
    _position.X -= 400f * (float)gameTime.ElapsedGameTime.TotalSeconds;
    _position.X = MathHelper.Clamp(_position.X, 0, float.MaxValue);
    _isRunning = true;         
}

if (state.IsKeyDown(Keys.Space) && !_isJumping) {
    _isJumping = true;
    _jumpVelocity = -JumpHeight;
    _jumpSoundInstance.Play();
}

if (_isJumping) {
    _jumpVelocity += Gravity * (float)gameTime.ElapsedGameTime.TotalSeconds;
    _position.Y += _jumpVelocity * (float)gameTime.ElapsedGameTime.TotalSeconds;

    if (_position.Y >= 750) {
        _position.Y = 750;
        _isJumping = false;
    }
}

Overview

เกม VR จำลองอาชีพแห่งอนาคต พัฒนาให้กับ NSM เพื่อเปิดโอกาสให้ผู้เล่นได้เข้าไปสัมผัสประสบการณ์การทำงานในโลกเสมือนจริง โปรเจกต์นี้ท้าทายในด้านการสร้างระบบโต้ตอบ (Interactive) ที่ต้องทำให้ผู้เล่นรู้สึกว่ากำลังใช้งานเครื่องมืออาชีพนั้นๆ ได้อย่างเป็นธรรมชาติที่สุด

Key Features

  • VR Interactions: พัฒนาระบบ Interactive ในโลก VR (การหยิบจับ, การใช้งานเครื่องมือตามสายอาชีพ)
  • Job Simulation: สร้างระบบจำลองสถานการณ์อาชีพ (เช่น วิศวกรพลังงาน, เกษตรอัจฉริยะ) ให้ทำงานประสานกับ UI/UX
  • Immersion: วางระบบและเขียนโค้ดรองรับมิติเสียง (Spatial Audio) เพิ่มความสมจริงในโลกเสมือน

Challenges & Solutions

การจัดการ Performance บนอุปกรณ์ VR: ปัญหาหลักที่พบคือ Framerate ตกเมื่อมี Particle Effects หรือวัตถุจำนวนมากในฉาก ซึ่งทำให้ผู้เล่นเกิดอาการ Motion Sickness ได้ ผมได้แก้ไขปัญหานี้โดยการเขียนระบบ Object Pooling เพื่อนำวัตถุที่ใช้แล้วกลับมาใช้ใหม่ แทนการลบและสร้างใหม่ (Destroy/Instantiate) ตลอดเวลา และปรับแต่ง Shader ให้เหมาะสมกับอุปกรณ์ VR

ตัวอย่างโค้ด: Object Pooling

public class ObjectPool : MonoBehaviour {
    private Queue<GameObject> pool = new Queue<GameObject>();
    
    public void Initialize(GameObject prefab, int size) {
        for (int i = 0; i < size; i++) {
            GameObject obj = Instantiate(prefab);
            obj.SetActive(false);
            pool.Enqueue(obj);
        }
    }

    public GameObject GetFromPool() {
        if (pool.Count > 0) {
            GameObject obj = pool.Dequeue();
            obj.SetActive(true);
            return obj;
        }
        return null;
    }
}

ตัวอย่างโค้ด: VR Item Grabbing Event

นอกจากการจัดการ Performance แล้ว การทำให้วัตถุโต้ตอบกับมือผู้เล่นได้อย่างสมจริงก็สำคัญ ผมได้ใช้ Event-driven programming ผูกเข้ากับ XR Interaction Toolkit เพื่อรัน Logic ทันทีที่ผู้เล่นหยิบเครื่องมือขึ้นมา 

public class VREquipment : MonoBehaviour {
    private XRGrabInteractable grabInteractable;

    void Awake() {
        grabInteractable = GetComponent<XRGrabInteractable>();
        grabInteractable.selectEntered.AddListener(OnGrabbed);
        grabInteractable.selectExited.AddListener(OnReleased);
    }

    private void OnGrabbed(SelectEnterEventArgs args) {
        // รันอนิเมชันหรือเปิดการทำงานของอุปกรณ์เมื่อถูกหยิบ
        EnableToolFunctionality();
        HapticFeedback.TriggerVibration(args.interactorObject);
    }
}
